GitHub-ல் இருந்து தானாக இயங்கும் நவீன தளம்

கட்டுரைகளைத் தேடுகிறது...
(Loading from GitHub...)

const apiUrl = `https://api.github.com/repos/${GITHUB_USERNAME}/${GITHUB_REPO}/contents/${FOLDER_NAME}`; async function fetchBlogs() { const blogListContainer = document.getElementById('blog-list'); try { // GitHub API-ல் இருந்து கோப்புகளைத் தேடுதல் const response = await fetch(apiUrl); if (!response.ok) { throw new Error('GitHub-ல் இருந்து தரவைப் பெறுவதில் சிக்கல்!'); } const files = await response.json(); blogListContainer.innerHTML = ''; // Loading-ஐ நீக்குதல் // கோப்புகள் எதுவும் இல்லை என்றால் if (files.length === 0) { blogListContainer.innerHTML = '

வலைப்பதிவுகள் எதுவும் இல்லை!

'; return; } // ஒவ்வொரு ஃபைலாக எடுத்து Card உருவாக்குதல் files.forEach(file => { // .html அல்லது .md கோப்புகளை மட்டும் எடுத்துக்கொள்ளுதல் if (file.name.endsWith('.html') || file.name.endsWith('.md')) { // ஃபைல் பெயரில் உள்ள '-' மற்றும் நீட்டிப்புகளை நீக்கி Title-ஆக மாற்றுதல் // எ.கா: "my-first-post.html" -> "My First Post" let title = file.name.replace(/\.(html|md)$/, '').replace(/-/g, ' '); title = title.charAt(0).toUpperCase() + title.slice(1); // கார்டு (Card) உருவாக்குதல் const card = document.createElement('div'); card.className = 'blog-card'; card.innerHTML = `

${title}

இந்தக் கட்டுரை GitHub-ல் இருந்து தானாகவே எடுக்கப்பட்டது.

முழுமையாகப் படிக்க → `; blogListContainer.appendChild(card); } }); } catch (error) { console.error('Error:', error); blogListContainer.innerHTML = `

பிழை: ${error.message}
(GitHub Username மற்றும் Repo Name சரியாக உள்ளதா என சரிபார்க்கவும்)

`; } } // வலைத்தளம் திறந்தவுடன் இயங்க fetchBlogs();
